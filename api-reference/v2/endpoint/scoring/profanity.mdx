### Response properties

import V2EnvelopeSuccess from "/snippets/v2-envelope-success.mdx";
import V2EnvelopeExecutionMs from "/snippets/v2-envelope-execution-ms.mdx";

<V2EnvelopeSuccess />

<ResponseField name="data" type="object" required>
  <Expandable title="properties" defaultOpen={true}>
    <ResponseField name="risk_level" type="string" required>
      Overall profanity risk for the submitted text (`low`, `medium`, or `high`). Greip classifies matches into levels: the highest severity wins (risky phrases and words first, then medium-risk, then low-risk). A clean text returns `low`.
    </ResponseField>
    <ResponseField name="recommendation" type="string" required>
      Suggested action (`accept`, `review`, or `reject`) derived from `risk_level`.
    </ResponseField>
  </Expandable>
</ResponseField>
